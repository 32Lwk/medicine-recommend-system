% 開始

\node[process node, below=of check1, xshift=-3.5cm] (layer2) {Layer 2: NLU処理};
\node[process node, below=of layer2] (pattern_match) {パターンマッチング\\$\mathcal{D}_{sym}$};

\node[process node, below=of check2, xshift=-3.5cm] (layer3) {Layer 3: 禁忌チェック};

\node[process node, below=of check3, xshift=-3.5cm] (layer4) {Layer 4: スコアリング};
\node[process node, below=of layer4] (vector_sim) {$\text{Sim} = \frac{v_u \cdot v_m}{\|v_u\|\|v_m\|}$\\$v_u, v_m \in \mathbb{R}^{1536}$};
\node[process node, below=of vector_sim] (coverage) {$\text{Cov} = \frac{|S_u \cap E_m|}{|S_u|}$};
\node[process node, below=of coverage] (weighted) {$\alpha(\sigma) \cdot \text{Sim}$\\$+ \beta(\sigma) \cdot \text{Cov}$};
\node[process node, below=of weighted] (risk_penalty) {$P_{risk}(m) = w_{side} \cdot R_{side}$\\$+ w_{inter} \cdot R_{inter}$};
\node[process node, below=of risk_penalty] (final_score) {$S(m) = \alpha \cdot \text{Sim}$\\$+ \beta \cdot \text{Cov} - P_{risk}$};
\node[decision node, below=of final_score] (check4) {アレルギー\\検出};

% Layer 5: 最終判定（詳細）
\node[process node, below=of check4, xshift=-3.5cm] (layer5) {Layer 5: 最終判定};
\node[process node, below=of layer5] (sort) {スコア順ソート};
\node[process node, below=of sort] (topk) {$top_k = 3$選定};
